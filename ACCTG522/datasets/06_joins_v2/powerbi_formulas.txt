LocationCode =
VAR Code = TRIM('EmployeeData'[EmployeeCode])
VAR FirstDigit = MINX(
    FILTER(GENERATESERIES(1, LEN(Code)),
        VAR Ch = MID(Code, [Value], 1)
        RETURN Ch >= "0" && Ch <= "9"), [Value])
VAR Raw = IF(ISBLANK(FirstDigit), BLANK(), UPPER(LEFT(Code, FirstDigit - 1)))
RETURN SWITCH(Raw,
    "JAPAN", "JAP", "CANADA", "CAN", "US", "USA",
    "AUS", "AUS", "GER", "GER", 
    "ENG", "ENG", "ARG", "ARG", "MEX", "MEX", BLANK())

##################################################################################################

EmpID =
VAR Code = 'EmployeeData'[EmployeeCode]
VAR Dash = SEARCH("-", Code, 1, 0)
VAR StartPos = MINX(
    FILTER(GENERATESERIES(1, LEN(Code)),
        VAR Ch = MID(Code, [Value], 1)
        RETURN Ch >= "0" && Ch <= "9"), [Value])
RETURN IF(NOT ISBLANK(StartPos) && Dash > StartPos,
    MID(Code, StartPos, Dash - StartPos), BLANK())

##################################################################################################

PlantID =
VAR Code = 'EmployeeData'[EmployeeCode]
VAR DashPosition = SEARCH("-", Code, 1, 0)
VAR CodeTail =
    IF(
        DashPosition > 0,
        MID(Code, DashPosition + 1, LEN(Code)),
        ""
    )
VAR LetterPos =
    MINX(
        FILTER(
            GENERATESERIES(1, LEN(CodeTail)),
            VAR Ch = MID(CodeTail, [Value], 1)
            RETURN NOT (Ch >= "0" && Ch <= "9")
        ),
        [Value]
    )
RETURN
    IF(
        LetterPos > 1,
        LEFT(CodeTail, LetterPos - 1),
        BLANK()
    )

##################################################################################################

PayCode =
VAR Code = 'EmployeeData'[EmployeeCode]
VAR DashPosition = SEARCH("-", Code, 1, 0)
VAR CodeTail =
    IF(
        DashPosition > 0,
        MID(Code, DashPosition + 1, LEN(Code)),
        ""
    )
VAR PlantLength = LEN('EmployeeData'[PlantID])
RETURN
    IF(
        LEN(CodeTail) > PlantLength,
        MID(CodeTail, PlantLength + 1, LEN(CodeTail)),
        BLANK()
    )


##################################################################################################

ActualPayFrequency =
SWITCH(UPPER('EmployeeData'[PayCode]),
    "W", "Weekly", "M", "Monthly", "MO", "Monthly", BLANK())


##################################################################################################

CountryPlantID =
IF(NOT ISBLANK('EmployeeData'[LocationCode]) &&
   NOT ISBLANK('EmployeeData'[PlantID]),
   'EmployeeData'[LocationCode] & "-" & 'EmployeeData'[PlantID], BLANK())


##################################################################################################

PlantType =
LOOKUPVALUE('PlantLookup'[PlantType],
    'PlantLookup'[CountryPlantID], 'EmployeeData'[CountryPlantID])


##################################################################################################

ExpectedPayFrequency =
LOOKUPVALUE('PlantLookup'[ExpectedPayFrequency],
    'PlantLookup'[CountryPlantID], 'EmployeeData'[CountryPlantID])

##################################################################################################

ExceptionFlag =
IF(ISBLANK('EmployeeData'[ActualPayFrequency]) ||
   ISBLANK('EmployeeData'[ExpectedPayFrequency]),
   "CHECK / UNMATCHED",
   IF('EmployeeData'[ActualPayFrequency] <>
      'EmployeeData'[ExpectedPayFrequency], "EXCEPTION", "OK"))
